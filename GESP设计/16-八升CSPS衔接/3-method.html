2 · KMP next 数组 
void getNext(const string& p) {
    nx[0] = 0;
    for (int i=1, k=0; i < p.size(); ++i) {
        while (k && p[k] != p[i]) k = nx[k-1];
        if (p[k] == p[i]) ++k;
        nx[i] = k;
    }
}
关键 · next[i] = P[0..i] 的最长公共前后缀长度

int fa[N];
int find(int x) {
    return fa[x] == x ? x : fa[x] = find(fa[x]);

void merge(int x, int y) {
    fa[find(x)] = find(y);
}
