2 · Dijkstra 堆优化 
priority_queue<pii, vector<pii>, greater<>> pq;
pq.push({0, s}); d[s] = 0;
while (!pq.empty()) {
    auto [dv,u] = pq.top(); pq.pop();
    if (dv > d[u]) continue;
    for (auto [v,w] : adj[u])
        if (d[u]+w < d[v]) {
            d[v] = d[u]+w;
            pq.push({d[v], v});
        }
}
关键 · 不能处理负权 · O((n+m) log n)
3 · Floyd 多源最短路 
for (int k = 1; k <= n; k++)        // k 必须最外层

void dfs(int u, int p) {
    dp[u][0] = 0;
    dp[u][1] = w[u];
    for (int v : adj[u]) if (v != p) {
        dfs(v, u);
        dp[u][0] += max(dp[v][0], dp[v][1]);
        dp[u][1] += dp[v][0];
    }
}
