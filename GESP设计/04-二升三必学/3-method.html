string toBase(int n, int r) {
    if (n == 0) return "0";
    string s;
    while (n) {
        int d = n % r;
        s += (d < 10 ? char('0'+d) : char('A'+d-10));
        n /= r;
    }
    reverse(s.begin(), s.end());
    return s;
}

int popcount(int x) {
    int c = 0;
    while (x) { c += x & 1; x >>= 1; }
    return c;
}
