q.push(s); vis[s] = true;
while (!q.empty()) {
    int u = q.front(); q.pop();
    for (int v : adj[u])
        if (!vis[v]) { vis[v]=true; q.push(v); }
}

void dfs(int u) {
    vis[u] = true;
    for (int v : adj[u])
        if (!vis[v]) dfs(v);
}

3 · 哈希表（链地址法） 
vector<int> ht[P];
void insert(int x) { ht[x % P].push_back(x); }
bool find(int x) {
    for (int v : ht[x % P])
        if (v == x) return true;
    return false;
}
