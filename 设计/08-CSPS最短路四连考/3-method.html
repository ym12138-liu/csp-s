分层 Dijkstra · 核心转移 
struct State { ll d; int u, used; };
priority_queue<State, ..., greater<>> pq;
pq.push({0, s, 0});  d[s][0] = 0;

while (!pq.empty()) {
    auto [d_, u, used] = pq.top(); pq.pop();
    if (d_ > d[u][used]) continue;
    for (auto [v, w] : adj[u]) {
        // 1. 普通转移：层不变
        if (d[u][used] + w < d[v][used]) ...;
        // 2. 用免费边：层从 0 跳到 1
        if (used == 0 && d[u][0] < d[v][1]) ...;
    }
}
