struct State { int d, u, used; };
priority_queue<State, vector<State>, greater<>> pq;
pq.push({0, s, 0}); d[s][0] = 0;
while (!pq.empty()) {
  auto [dv, u, used] = pq.top(); pq.pop();
  // 1. 普通转移层不变
  if (d[u][used] + w < d[v][used]) ...
  // 2. 用免费边层从 0 跳到 1
  if (used == 0 && d[u][0] < d[v][1]) ...
}
