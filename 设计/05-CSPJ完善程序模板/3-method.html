1 · 二分查找（lower_bound） 
int lb(int* a, int n, int x) {
    int l = 0, r = n;
    while (l < r) {
        int mid = (l + r) >> 1;
        if (a[mid] < x) l = mid + 1;
        else r = mid;
    }
    return l;
}
关键 · 半开区间 [l, r)；l = mid+1, r = mid

    else if (a[i] == cand) cnt++;
    else cnt--;
